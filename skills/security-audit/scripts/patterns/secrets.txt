# Patterns for hardcoded secrets / credentials in source.
# Used by 07-secrets.sh (grep -Ei). Fallback layer — runs even without gitleaks.
# Format: one extended-regex per line, '#' comments and blank lines ignored.

# ── Cloud provider keys ────────────────────────────────────────────────────────
AKIA[0-9A-Z]{16}
ASIA[0-9A-Z]{16}
AWS_SECRET_ACCESS_KEY\s*[:=]
AIza[0-9A-Za-z_\-]{35}
ya29\.[0-9A-Za-z_\-]+
AccountKey=[A-Za-z0-9+/=]{50,}

# ── Git host / CI tokens ───────────────────────────────────────────────────────
gh[pousr]_[A-Za-z0-9]{36}
github_pat_[A-Za-z0-9_]{60,}
glpat-[A-Za-z0-9_\-]{20}

# ── SaaS API keys ──────────────────────────────────────────────────────────────
xox[baprs]-[A-Za-z0-9-]{10,}
sk_live_[A-Za-z0-9]{20,}
rk_live_[A-Za-z0-9]{20,}
SG\.[A-Za-z0-9_\-]{22}\.[A-Za-z0-9_\-]{43}
sk-[A-Za-z0-9]{20,}
sk-ant-[A-Za-z0-9_\-]{20,}
ntn_[A-Za-z0-9]{40,}
secret_[A-Za-z0-9]{40,}
dop_v1_[a-f0-9]{64}
AC[a-z0-9]{32}

# ── Private keys ───────────────────────────────────────────────────────────────
-----BEGIN (RSA|EC|DSA|OPENSSH|PGP|ENCRYPTED)? ?PRIVATE KEY-----

# ── JWT ────────────────────────────────────────────────────────────────────────
eyJ[A-Za-z0-9_\-]{10,}\.eyJ[A-Za-z0-9_\-]{10,}\.[A-Za-z0-9_\-]{10,}

# ── Generic assignments (high-noise, kept last) ────────────────────────────────
(api[_-]?key|apikey|access[_-]?token|auth[_-]?token|client[_-]?secret|secret[_-]?key|password|passwd)\s*[:=]\s*['"][^'"]{12,}['"]
